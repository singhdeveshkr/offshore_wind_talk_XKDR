# Sampling frequency [Hz]
# Fs = 2 means we generate 2 data points every second
Fs = 2.0

# Total simulation time [seconds]
T = 1122.0

# Create time array:
# starts at 0 sec, ends before T, with spacing = 1/Fs
# Example: 0, 0.5, 1.0, 1.5 ...
t = np.arange(0, T, 1/Fs)


# ------------------------------------------------------------------
# Frequency components used to build the irregular wave signal
# ------------------------------------------------------------------

# Create 200 frequency components between 0.04 Hz and 0.30 Hz
# Lower frequency = long waves
# Higher frequency = short waves
freqs_comp = np.linspace(0.04, 0.30, 200)

# Peak frequency [Hz]
# This is the frequency where most wave energy is concentrated
fp = 0.10


# ------------------------------------------------------------------
# JONSWAP-type spectral shaping
# ------------------------------------------------------------------

# sigma controls the width of the spectral peak
# Lower side of peak uses 0.07
# Higher side of peak uses 0.09
# This asymmetry is common in ocean wave spectra
sigma = np.where(freqs_comp <= fp, 0.07, 0.09)

# Peak enhancement term
# Makes the spectrum sharper around the peak frequency
r = np.exp(-((freqs_comp - fp)**2) / (2 * sigma**2 * fp**2))

# Base spectral shape
# Similar to Pierson-Moskowitz/JONSWAP formulation
# freq^-5 term controls high-frequency decay
# exponential term controls low-frequency behaviour
S_comp = (freqs_comp**(-5) * np.exp(-1.25*(freqs_comp/fp)**(-4)))

# Normalize spectrum so maximum value becomes 1
# Easier for numerical scaling and visualization
S_comp /= S_comp.max()


# ------------------------------------------------------------------
# Convert spectral energy into wave amplitudes
# ------------------------------------------------------------------

# Amplitude corresponding to each frequency component
# sqrt(2*S/N) converts spectral density into sinusoidal amplitudes
amplitudes = np.sqrt(2 * S_comp / len(freqs_comp))

# Random phase assigned to each frequency component
# This randomness creates an irregular sea state
phases = np.random.uniform(0, 2*np.pi, len(freqs_comp))


# ------------------------------------------------------------------
# Construct irregular wave signal
# ------------------------------------------------------------------

# Initialize empty signal
X = np.zeros(len(t))

# Add sinusoidal components one-by-one
# Each component has:
#   a   -> amplitude
#   f   -> frequency
#   phi -> random phase
for a, f, phi in zip(amplitudes, freqs_comp, phases):

    # Sum all sinusoidal wave components
    X += a * np.sin(2*np.pi*f*t + phi)


# ------------------------------------------------------------------
# Introduce mild non-linearity
# ------------------------------------------------------------------

# Add quadratic contribution
# This creates slightly sharper crests and flatter troughs,
# which is more realistic for ocean waves
X = X + 0.18 * X**2 / np.std(X)

# Remove mean value so signal oscillates around zero
X -= np.mean(X)


# ------------------------------------------------------------------
# Final statistical parameter
# ------------------------------------------------------------------

# Standard deviation of generated signal
# Often related to wave energy or significant wave height scaling
sigma_X = np.std(X)

from scipy.signal import argrelextrema

# Find local maxima (wave crests)
# order=4 means point must be greater than 4 neighbouring points
idx_max = argrelextrema(X, np.greater, order=4)[0]

# Find local minima (wave troughs)
idx_min = argrelextrema(X, np.less, order=4)[0]

# Sort indices in increasing time order
idx_max = np.sort(idx_max)
idx_min = np.sort(idx_min)


# -------------------------------------------------------
# Pair each trough with the next crest
# -------------------------------------------------------

pairs = []
i, j = 0, 0

while i < len(idx_min) and j < len(idx_max):

    # Valid wave cycle:
    # trough occurs first, then crest
    if idx_min[i] < idx_max[j]:

        # Store pair of indices
        pairs.append((idx_min[i], idx_max[j]))

        i += 1
        j += 1

    else:
        # Skip crest if it occurs before trough
        j += 1

pairs = np.array(pairs)


# -------------------------------------------------------
# Keep only first 300 seconds for plotting
# -------------------------------------------------------

mask_max = t[idx_max] < 300
mask_min = t[idx_min] < 300

# Use trough time for filtering wave pairs
mask_pairs = t[pairs[:,0]] < 300

idx_max_f = idx_max[mask_max]
idx_min_f = idx_min[mask_min]
pairs_f   = pairs[mask_pairs]

Meaning of Crossing Intensity¶

Crossing intensity $\mu(u)$ represents:

$$ \mu(u) $$

the average rate at which the wave signal crosses a particular level $u$.

In this implementation, a crossing is counted when:

  • trough value $m < u$
  • crest value $M > u$

which means the wave cycle passes through the level $u$.

Physical Interpretation¶

If:

$$ \mu(1.0) = 0.02 $$

then the sea surface crosses the level of $1.0 \, m$ approximately:

  • $0.02$ times per second
  • or once every $50$ seconds on average.

Behaviour of the Curve¶

  • Near the mean water level ($u \approx 0$):

    • crossings are frequent
    • many waves pass through this level

  • At higher levels:

    • crossings become rare
    • only large waves exceed those elevations

Therefore, the crossing intensity curve decreases as $u$ increases.

Importance in Offshore Engineering¶

Crossing intensity is commonly used in:

  • extreme wave analysis
  • deck wetness assessment
  • slamming analysis
  • fatigue cycle estimation
  • reliability studies

It helps estimate:

"How often does the response exceed a certain level?"

# Trough values from each wave pair
m_vals = X[pairs[:,0]]

# Crest values from each wave pair
M_vals = X[pairs[:,1]]


# Create level range for evaluation
# u represents different elevation levels
u_grid = np.linspace(X.min(), X.max(), 200)


# -------------------------------------------------------
# Crossing intensity function
# -------------------------------------------------------

def crossing(u):

    # Count waves crossing level u
    # Condition:
    # trough below u AND crest above u
    return np.sum((m_vals < u) & (M_vals > u)) / T


# Compute crossing intensity for all levels
mu_hat = np.array([crossing(u) for u in u_grid])


# -------------------------------------------------------
# Smooth the curve
# -------------------------------------------------------

# Moving average smoothing
# Helps remove small numerical fluctuations
mu_hat = np.convolve(mu_hat, np.ones(5)/5, mode='same')


# -------------------------------------------------------
# Enforce physical behaviour
# -------------------------------------------------------

# Crossing intensity should decrease
# as level becomes larger
mu_hat = np.maximum.accumulate(mu_hat[::-1])[::-1]


# -------------------------------------------------------
# Gaussian reference model
# -------------------------------------------------------

# Standard deviation of signal
sigma_X = np.std(X)

# Peak crossing intensity
mu_0 = mu_hat.max()

# Ideal Gaussian crossing curve
mu_gauss = mu_0 * np.exp(-u_grid**2 / (2 * sigma_X**2))

# Plot results

plt.figure(figsize=(8,4))

Transformation Function¶

The transformation function $g(u)$ converts the original non-Gaussian wave elevation into an equivalent Gaussian variable.

$$ g(u) $$

The idea is:

  • real ocean waves are often nonlinear and non-Gaussian
  • Gaussian processes are mathematically easier to analyze
  • therefore, a transformation is used to map the measured signal into Gaussian space

In this method, the transformation is derived from the crossing intensity function $\mu(u)$.

If the transformed curve follows a straight Gaussian reference line, the process behaves close to Gaussian.

Deviations from the Gaussian line indicate nonlinear wave behaviour such as:

  • sharper crests
  • flatter troughs
  • asymmetry in wave shape

# Frequency values corresponding to FFT components
# rfftfreq is used because signal is real-valued
f_fft = np.fft.rfftfreq(len(X), d=1/Fs)

# Fast Fourier Transform of signal
# Converts time-domain signal into frequency-domain
X_fft = np.fft.rfft(X)


# -------------------------------------------------------
# Power Spectral Density (PSD)
# -------------------------------------------------------

# PSD represents how wave energy is distributed
# across different frequencies
S = (np.abs(X_fft)**2) / (len(X) * Fs)


# -------------------------------------------------------
# Smooth spectrum
# -------------------------------------------------------

# Moving-average smoothing function
# Helps remove noisy spikes from FFT result
def smooth(arr, window=40):

    kernel = np.ones(window) / window

    return np.convolve(arr, kernel, mode='same')


# Smoothed spectral density
S_smooth = smooth(S, 60)


# -------------------------------------------------------

# -------------------------------------------------------
# JONSWAP Spectrum Function
# -------------------------------------------------------

# JONSWAP is a commonly used ocean wave spectrum
# for developing sea states

def jonswap(f, fp=0.10, gamma=3.3):

    # Avoid division by zero at f = 0
    f = np.clip(f, 1e-6, None)

    # Spectral width parameter
    # Different values below and above peak frequency
    sigma = np.where(f <= fp, 0.07, 0.09)

    # Peak enhancement term
    # Controls sharpness around peak frequency
    r = np.exp(-((f - fp)**2) / (2 * sigma**2 * fp**2))

    # JONSWAP spectral equation
    S = (f**(-5) *
         np.exp(-1.25*(f/fp)**(-4)) *
         gamma**r)

    # Normalize spectrum
    S /= np.max(S)

    # Scale to match estimated spectrum magnitude
    return S * np.max(S_smooth)


# Compute theoretical JONSWAP spectrum
S_j = jonswap(f_fft)


# -------------------------------------------------------
# Plot JONSWAP reference
# -------------------------------------------------------

# Compare theoretical spectrum
# against estimated spectrum
plt.plot(f_fft[mask],
         S_j[mask],
         '--',
         label="JONSWAP",
         color="orange")

# -------------------------------------------------------
# Step 9 — Extreme Wave Prediction (CORRECTED)
# -------------------------------------------------------

# Standard deviation of wave elevation
sigma_X = np.std(X)

# -------------------------------------------------------
# Crest height range for prediction
# -------------------------------------------------------

# Extended range for extreme extrapolation
u_ext = np.linspace(
    0.5 * sigma_X,
    10 * sigma_X,
    5000
)

# -------------------------------------------------------
# Spectral moments
# -------------------------------------------------------

m0 = np.trapz(S_smooth, f_fft)

m2 = np.trapz(
    (2*np.pi*f_fft)**2 * S_smooth,
    f_fft
)

# -------------------------------------------------------
# Mean zero-upcrossing frequency
# -------------------------------------------------------

nu0 = (1/(2*np.pi)) * np.sqrt(m2/m0)

# -------------------------------------------------------
# Gaussian exceedance intensity
# -------------------------------------------------------
#
# Rayleigh exceedance model for crest amplitudes
#
# P(C > u) = exp(-u² / (2σ²))
#
# Crossing intensity:
#
# μ(u) = ν0 * P(C > u)
#
# -------------------------------------------------------

mu_G = nu0 * np.exp(
    -(u_ext**2) / (2 * sigma_X**2)
)

# -------------------------------------------------------
# Non-Gaussian correction
# -------------------------------------------------------

nG_factor = (
    1
    + 0.6 *
    np.clip(
        (u_ext/sigma_X - 1),
        0,
        None
    )**1.7
)

mu_NG = mu_G * nG_factor

# -------------------------------------------------------
# Return period
# -------------------------------------------------------

sec_per_year = 365.25 * 24 * 3600

# events/sec → years/event
Tr_G = 1 / (mu_G * sec_per_year)

Tr_NG = 1 / (mu_NG * sec_per_year)

# -------------------------------------------------------
# VERY IMPORTANT FIX
# -------------------------------------------------------
#
# np.interp REQUIRES ascending x-array
#
# Return periods naturally become:
#
# small crest → small Tr
# large crest → huge Tr
#
# but numerical noise may break monotonicity
#
# so explicitly sort them
#
# -------------------------------------------------------

idxG = np.argsort(Tr_G)
idxNG = np.argsort(Tr_NG)

Tr_G_sorted = Tr_G[idxG]
u_G_sorted = u_ext[idxG]

Tr_NG_sorted = Tr_NG[idxNG]
u_NG_sorted = u_ext[idxNG]

# -------------------------------------------------------
# Design waves
# -------------------------------------------------------

u10_G = np.interp(
    10,
    Tr_G_sorted,
    u_G_sorted
)

u100_G = np.interp(
    100,
    Tr_G_sorted,
    u_G_sorted
)

u1000_G = np.interp(
    1000,
    Tr_G_sorted,
    u_G_sorted
)

u10_NG = np.interp(
    10,
    Tr_NG_sorted,
    u_NG_sorted
)

u100_NG = np.interp(
    100,
    Tr_NG_sorted,
    u_NG_sorted
)

u1000_NG = np.interp(
    1000,
    Tr_NG_sorted,
    u_NG_sorted
)

# -------------------------------------------------------
# Plot

plt.figure(figsize=(11,6))

plt.semilogy(
